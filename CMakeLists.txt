cmake_minimum_required(VERSION 3.20)
project(ArchipelagoRAC2 LANGUAGES C CXX)
if(DEFINED OPENRAC_SDK)
  set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
endif()
find_package(OpenRACMod 1.19 REQUIRED)
openrac_add_mod(aprac2_probe MANIFEST probe/openrac-mod.json SOURCES src/probe.c)
set_target_properties(aprac2_probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
if(WIN32)
  openrac_add_mod(aprac2_client MANIFEST openrac-mod.json
    SOURCES src/development_client.cpp src/bridge.cpp src/runtime.cpp)
  target_include_directories(aprac2_client PRIVATE third_party)
  target_compile_features(aprac2_client PRIVATE cxx_std_17)
  target_link_libraries(aprac2_client PRIVATE ws2_32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(probe_test tests/probe_test.cpp)
  target_link_libraries(probe_test PRIVATE OpenRAC::mod)
  target_compile_features(probe_test PRIVATE cxx_std_17)
  add_test(NAME probe_host COMMAND probe_test)
  add_executable(journal_test tests/journal_test.cpp)
  target_include_directories(journal_test PRIVATE third_party src)
  target_compile_features(journal_test PRIVATE cxx_std_17)
  add_test(NAME save_journal COMMAND journal_test)
  if(WIN32)
    add_executable(bridge_test tests/bridge_test.cpp src/bridge.cpp)
    target_include_directories(bridge_test PRIVATE third_party src)
    target_compile_features(bridge_test PRIVATE cxx_std_17)
    target_link_libraries(bridge_test PRIVATE ws2_32)
    add_test(NAME native_bridge COMMAND bridge_test)
    add_executable(runtime_test tests/runtime_test.cpp src/runtime.cpp src/bridge.cpp)
    target_include_directories(runtime_test PRIVATE third_party src)
    target_link_libraries(runtime_test PRIVATE OpenRAC::mod ws2_32)
    target_compile_features(runtime_test PRIVATE cxx_std_17)
    add_test(NAME native_runtime COMMAND runtime_test)
  endif()
endif()
