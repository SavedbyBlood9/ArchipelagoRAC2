cmake_minimum_required(VERSION 3.20)
project(ArchipelagoRAC2 LANGUAGES C CXX)
# The SDK: OPENRAC_SDK=<dir> (an unpacked SDK, or sdk/ in a checkout of OpenRAC), or the release
# sdk.lock pins (python tools/fetch_sdk.py puts it in .sdk/).
if(NOT OpenRACMod_DIR)
  if(DEFINED OPENRAC_SDK)
    set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
  else()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" lock)
    string(JSON sdk_version GET "${lock}" version)
    set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
    if(NOT EXISTS "${OpenRACMod_DIR}/OpenRACModConfig.cmake")
      message(FATAL_ERROR "No OpenRAC mod SDK ${sdk_version}: run python tools/fetch_sdk.py first "
                          "(or pass -DOPENRAC_SDK=<an unpacked SDK>)")
    endif()
  endif()
endif()
find_package(OpenRACMod 1.40 REQUIRED)
find_package(Threads REQUIRED)

# The development probe is a diagnostic, not part of a release: -DAPRAC2_PROBE=ON builds it.
option(APRAC2_PROBE "Build the aprac2-openrac-probe diagnostic mod" OFF)
if(APRAC2_PROBE)
  openrac_add_mod(aprac2_probe MANIFEST probe/openrac-mod.json SOURCES src/probe.c)
  set_target_properties(aprac2_probe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
endif()

# The client: Winsock on Windows, POSIX sockets elsewhere (src/net.hpp).
set(APRAC2_SOCKETS Threads::Threads)
if(WIN32)
  list(APPEND APRAC2_SOCKETS ws2_32)
endif()
openrac_add_mod(aprac2_client MANIFEST openrac-mod.json
  SOURCES src/development_client.cpp src/bridge.cpp src/runtime.cpp)
target_include_directories(aprac2_client PRIVATE third_party)
target_compile_features(aprac2_client PRIVATE cxx_std_17)
target_link_libraries(aprac2_client PRIVATE ${APRAC2_SOCKETS})
include(CTest)
if(BUILD_TESTING)
  add_executable(probe_test tests/probe_test.cpp)
  target_link_libraries(probe_test PRIVATE OpenRAC::mod)
  target_compile_features(probe_test PRIVATE cxx_std_17)
  add_test(NAME probe_host COMMAND probe_test)
  add_executable(journal_test tests/journal_test.cpp)
  target_include_directories(journal_test PRIVATE third_party src)
  target_compile_features(journal_test PRIVATE cxx_std_17)
  add_test(NAME save_journal COMMAND journal_test)
  add_executable(bridge_test tests/bridge_test.cpp src/bridge.cpp)
  target_include_directories(bridge_test PRIVATE third_party src)
  target_compile_features(bridge_test PRIVATE cxx_std_17)
  target_link_libraries(bridge_test PRIVATE ${APRAC2_SOCKETS})
  add_test(NAME native_bridge COMMAND bridge_test)
  add_executable(runtime_test tests/runtime_test.cpp src/runtime.cpp src/bridge.cpp)
  target_include_directories(runtime_test PRIVATE third_party src)
  target_link_libraries(runtime_test PRIVATE OpenRAC::mod ${APRAC2_SOCKETS})
  target_compile_features(runtime_test PRIVATE cxx_std_17)
  add_test(NAME native_runtime COMMAND runtime_test)
  add_test(NAME native_runtime_pal COMMAND runtime_test --pal)
  add_test(NAME native_runtime_ntsc2 COMMAND runtime_test --ntsc2)
  add_test(NAME native_runtime_options_off COMMAND runtime_test --ntsc --default-options)
  add_test(NAME native_runtime_pal_options_off COMMAND runtime_test --pal --default-options)
  add_test(NAME native_runtime_ntsc2_options_off COMMAND runtime_test --ntsc2 --default-options)
endif()
